1. BASE DE DATOS MYSQL
====================================================

CREATE DATABASE glunova_ai;

USE glunova_ai;

----------------------------------------------------
TABLA ROLES
----------------------------------------------------

CREATE TABLE roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre_rol VARCHAR(50)
);

INSERT INTO roles(nombre_rol) VALUES
('Administrador'),
('Medico'),
('Paciente');

----------------------------------------------------
TABLA USUARIOS
----------------------------------------------------

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50),
    password VARCHAR(255),
    rol_id INT,
    FOREIGN KEY (rol_id) REFERENCES roles(id)
);

----------------------------------------------------
TABLA PERMISOS
----------------------------------------------------

CREATE TABLE permisos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    rol_id INT,
    modulo VARCHAR(50),
    ver BOOLEAN,
    crear BOOLEAN,
    editar BOOLEAN,
    eliminar BOOLEAN,
    FOREIGN KEY (rol_id) REFERENCES roles(id)
);

----------------------------------------------------
TABLA PACIENTES
----------------------------------------------------

CREATE TABLE pacientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    documento VARCHAR(50),
    edad INT,
    tipo_diabetes VARCHAR(50),
    telefono VARCHAR(50),
    correo VARCHAR(100)
);

----------------------------------------------------
TABLA CONTROLES GLUCOSA
----------------------------------------------------

CREATE TABLE controles_glucosa (
    id INT AUTO_INCREMENT PRIMARY KEY,
    paciente_id INT,
    nivel_glucosa DECIMAL(10,2),
    fecha DATE,
    observaciones TEXT,
    FOREIGN KEY (paciente_id) REFERENCES pacientes(id)
);

5. CRUD PACIENTES
Archivo: pacientes.php
====================================================

<?php
include 'conexion.php';

if(isset($_POST['guardar'])){

    $nombre = $_POST['nombre'];
    $documento = $_POST['documento'];
    $edad = $_POST['edad'];
    $tipo = $_POST['tipo_diabetes'];
    $telefono = $_POST['telefono'];
    $correo = $_POST['correo'];

    $sql = "INSERT INTO pacientes
    (nombre, documento, edad, tipo_diabetes, telefono, correo)

    VALUES
    ('$nombre','$documento','$edad',
    '$tipo','$telefono','$correo')";

    $conn->query($sql);
}

if(isset($_GET['eliminar'])){

    $id = $_GET['eliminar'];

    $conn->query("DELETE FROM pacientes WHERE id=$id");
}
?>
